/* Holding Kashani V18.2 — character/avatar feature removed completely */
(function(){
'use strict';

function stripAvatarUi(root=document){
  try{
    root.querySelectorAll(
      '.avatar,.avatarShell,.avatarSvg,.avatar3dModel,.avatar3dWrap,.avatarStudio,.avatarStudioPreview,.avatarStudioControls,.avatarChoices,.avatarChoice,.avatarFeatureTabs,.avatarFeaturePanel,.avatarThumbGrid,.avatarPalette'
    ).forEach(el=>el.remove());
    root.querySelectorAll('[id*="avatar"],[id*="Avatar"]').forEach(el=>{
      if(el.id!=='') el.remove();
    });
    root.querySelectorAll('[onclick]').forEach(el=>{
      const oc=el.getAttribute('onclick')||'';
      if(/Avatar|avatar/.test(oc)) el.remove();
    });
  }catch(_){}
}

/* Remove stored character configuration from the client state. */
try{
  if(typeof state!=='undefined'){
    state.avatars={};
    if(typeof STORAGE_KEY!=='undefined') localStorage.setItem(STORAGE_KEY,JSON.stringify(state));
  }
}catch(_){}

/* Any legacy renderer that still asks for an avatar receives no markup. */
avatarSvg=function(){return ''};
openAvatarCustomizer=function(){if(typeof toast==='function')toast('بخش کاراکتر از برنامه حذف شده است')};

/* Settings without character controls. */
window.openV18Settings=function(){
  modal('تنظیمات',`<div class="settingsRows">
    <div class="settingsRow"><div><strong>ظاهر اپ</strong><small>حالت روشن / تیره برای همین حساب</small></div><button class="chip" onclick="toggleTheme();openV18Settings()">${currentTheme()==='dark'?'تیره':'روشن'}</button></div>
    ${isInternal()?`<div class="settingsRow"><div><strong>نیروها و دسترسی</strong><small>حساب، سابقه، آرشیو و رمزها</small></div><button class="chip" onclick="openEmployeeSettings()">مدیریت</button></div>
    <div class="settingsRow"><div><strong>دوره‌ها و قیمت‌ها</strong><small>فعال‌سازی و قیمت پایه</small></div><button class="chip" onclick="openCourseSettings()">مدیریت</button></div>`:''}
    <div class="settingsRow"><div><strong>اعلان‌ها</strong><small>پیام‌های مدیریت و Taskها</small></div><button class="chip" onclick="openV18Inbox()">مشاهده</button></div>
    <div class="settingsRow"><div><strong>گروه‌های تیم</strong><small>گفت‌وگوی داخلی تیم‌ها</small></div><button class="chip" onclick="openV18Groups()">گروه‌ها</button></div>
  </div>`);
  stripAvatarUi(document);
};

const _mountNoAvatar=mountApp;
mountApp=function(){_mountNoAvatar();stripAvatarUi(document)};

const _homeNoAvatar=home;
home=function(){_homeNoAvatar();stripAvatarUi($('#view')||document)};

const _rewardsNoAvatar=rewards;
rewards=function(){_rewardsNoAvatar();stripAvatarUi($('#view')||document)};

const _moreNoAvatar=more;
more=function(){_moreNoAvatar();stripAvatarUi($('#view')||document)};

const _renderNoAvatar=renderPage;
renderPage=function(name){_renderNoAvatar(name);stripAvatarUi(document)};
page=function(name){renderPage(name)};

stripAvatarUi(document);
})();
