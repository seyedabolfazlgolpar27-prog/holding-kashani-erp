addEmployee=async function(){const name=$('#emp_name').value.trim(),business_unit=$('#emp_division').value,role=$('#emp_role').value,username=$('#emp_username').value.trim(),password=$('#emp_password').value;if(!name||!username||!password)return toast('اطلاعات را کامل کن');if(password.length<6)return toast('رمز حداقل ۶ کاراکتر باشد');try{await api('/api/v18/users',{method:'POST',body:JSON.stringify({full_name:name,business_unit,role,username,password})});await refreshEmployees();renderEmployeeManager('active');toast('حساب جدید ساخته شد')}catch(e){toast(e.message==='username_reserved_for_history'?'این نام کاربری متعلق به یک هویت قبلی است؛ نام جدید انتخاب کن':e.message==='username_exists'?'نام کاربری تکراری است':'ساخت حساب انجام نشد')}};
window.manageEmployeeV18=function(id){const e=v18Employees.find(x=>Number(x.id)===Number(id));if(!e)return;if(!e.active){modal('بازگشت '+esc(e.full_name),`<div class="info">سوابق قبلی حذف نمی‌شود و یک دوره همکاری جدید به همان پرونده اضافه می‌شود.</div><button class="btn" style="margin-top:10px" onclick="reactivateEmployeeV18(${id})">فعال‌سازی مجدد</button>`);return}modal('مدیریت '+esc(e.full_name),`<div class="field"><label>رمز جدید (اختیاری)</label><input id="v18_new_pass" type="password"></div><button class="btn ghost" onclick="changeEmployeePassV18(${id})">تغییر رمز</button><div class="divider"></div><div class="field"><label>علت پایان همکاری</label><input id="v18_archive_reason" placeholder="مثلاً پایان قرارداد"></div><button class="btn alt" onclick="archiveEmployeeV18(${id})">پایان همکاری و انتقال به آرشیو</button><div class="notice" style="margin-top:9px">هیچ گزارش، فروش، Task یا سابقه‌ای پاک نمی‌شود.</div>`)};
window.changeEmployeePassV18=async function(id){const password=$('#v18_new_pass').value;if(password.length<6)return toast('رمز حداقل ۶ کاراکتر');try{await api('/api/v18/users/'+id,{method:'PATCH',body:JSON.stringify({password})});closeModal();toast('رمز تغییر کرد')}catch(_){toast('خطا در تغییر رمز')}};
window.archiveEmployeeV18=async function(id){const reason=$('#v18_archive_reason').value.trim()||'پایان همکاری';try{await api('/api/v18/users/'+id,{method:'PATCH',body:JSON.stringify({active:false,reason})});await refreshEmployees();renderEmployeeManager('archive');toast('نیرو آرشیو شد؛ داده‌ها حفظ شدند')}catch(_){toast('آرشیو انجام نشد')}};
window.reactivateEmployeeV18=async function(id){try{await api('/api/v18/users/'+id,{method:'PATCH',body:JSON.stringify({active:true})});await refreshEmployees();renderEmployeeManager('active');toast('دوره همکاری جدید شروع شد')}catch(_){toast('فعال‌سازی انجام نشد')}};
window.openEmployeeHistory=async function(id){try{const d=await api('/api/v18/users/'+id+'/history'),u=d.user,uid=d.state_user_id,months=monthOptions(12).reverse(),scores=months.map(k=>{try{return coinScoreForMonth(uid,k)}catch(_){return 0}}),mx=100,vals=scores.filter(x=>x>0),first=vals[0]||0,last=vals[vals.length-1]||0,delta=last-first;modal('پرونده '+esc(u.full_name),`<div class="card hero"><div class="muted">سابقه کل همکاری</div><div style="font-size:23px;font-weight:800">${daysToText(d.tenure_days)}</div><div class="muted">${u.active?'فعال':'آرشیو'} · ${esc(roleNames[u.role]||u.role)}</div></div><div class="card"><div class="row between"><h3 class="title" style="margin:0">روند ۱۲ ماه</h3><span class="${delta>=0?'ok':'danger'}">${delta>=0?'↗':'↘'} ${fmt(Math.abs(delta))}</span></div><div class="historyBars">${months.map((k,i)=>`<div class="historyBar" style="height:${Math.max(3,scores[i])}%"><b>${scores[i]||''}</b><small>${pMonthLabel(k).split(' ')[0].slice(0,3)}</small></div>`).join('')}</div><div class="muted" style="margin-top:23px">روند بر اساس امتیاز عملکرد ثبت‌شده در هر ماه محاسبه می‌شود.</div></div><div class="card"><h3 class="title">دوره‌های همکاری</h3>${d.periods.map((p,i)=>`<div class="item"><strong>دوره ${fmt(i+1)}</strong><small>${faDate(p.start_date)} تا ${p.end_date?faDate(p.end_date):'اکنون'}${p.reason?' · '+esc(p.reason):''}</small></div>`).join('')}</div>`)}catch(e){toast('دریافت سابقه انجام نشد')}};
