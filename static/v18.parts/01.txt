/* Keep archived identities in memory so historical reports never lose their owner. */
allUsers=function(){
  const boot=baseUsers.filter(x=>x.username===BOOT).map(x=>({...x,active:true,serverId:v18Employees.find(e=>e.username===BOOT)?.id}));

};

async function refreshEmployees(){
  if(!apiToken||!isManager())return;
  try{
    const d=await api('/api/v18/users');v18Employees=d.users||[];
    const old=new Map((state.customUsers||[]).map(x=>[Number(x.serverId||-1),x]));
    const me=await api('/api/me').catch(()=>null);
    const rows=v18Employees.filter(e=>e.username!==BOOT || e.id!==me?.id);
    const next=rows.map(e=>{const prev=old.get(Number(e.id))||{};return {...prev,id:prev.id||`custom_server_${e.id}`,serverId:e.id,division:e.business_unit,role:e.role,name:e.full_name,username:e.username,active:!!e.active,tenureDays:Number(e.tenure_days||0),archivedAt:e.archived_at||'',archiveReason:e.archive_reason||''}});
    state.customUsers=next;localStorage.setItem(STORAGE_KEY,JSON.stringify(state));
  }catch(e){console.warn('employee sync',e)}
}
