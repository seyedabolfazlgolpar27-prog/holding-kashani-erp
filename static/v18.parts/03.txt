function div(a,b){return ~~(a/b)}
function jalCal(jy){const breaks=[-61,9,38,199,426,686,756,818,1111,1181,1210,1635,2060,2097,2192,2262,2324,2394,2456,3178];let bl=breaks.length,gy=jy+621,leapJ=-14,jp=breaks[0],jm,jump,leap,leapG,march,n;if(jy<jp||jy>=breaks[bl-1])throw Error('Invalid Jalaali year '+jy);for(let i=1;i<bl;i+=1){jm=breaks[i];jump=jm-jp;if(jy<jm)break;leapJ=leapJ+div(jump,33)*8+div((jump%33),4);jp=jm}n=jy-jp;leapJ=leapJ+div(n,33)*8+div(((n%33)+3),4);if((jump%33)===4&&jump-n===4)leapJ+=1;leapG=div(gy,4)-div((div(gy,100)+1)*3,4)-150;march=20+leapJ-leapG;if(jump-n<6)n=n-jump+div(jump+4,33)*33;leap=((n+1)%33-1)%4;if(leap===-1)leap=4;return{leap,gy,march}}
function g2d(gy,gm,gd){let d=div((gy+div(gm-8,6)+100100)*1461,4)+div(153*((gm+9)%12)+2,5)+gd-34840408;d=d-div(div(gy+100100+div(gm-8,6),100)*3,4)+752;return d}
function d2g(jdn){let j=4*jdn+139361631;j=j+div(div(4*jdn+183187720,146097)*3,4)*4-3908;let i=div((j%1461),4)*5+308;let gd=div((i%153),5)+1;let gm=(div(i,153)%12)+1;let gy=div(j,1461)-100100+div(8-gm,6);return{gy,gm,gd}}
function j2d(jy,jm,jd){let r=jalCal(jy);return g2d(r.gy,3,r.march)+(jm-1)*31-div(jm,7)*(jm-7)+jd-1}
function d2j(jdn){let g=d2g(jdn),jy=g.gy-621,r=jalCal(jy),jdn1f=g2d(g.gy,3,r.march),k=jdn-jdn1f,jm,jd;if(k>=0){if(k<=185){jm=1+div(k,31);jd=(k%31)+1;return{jy,jm,jd}}k-=186}else{jy-=1;k+=179;if(r.leap===1)k+=1}jm=7+div(k,30);jd=(k%30)+1;return{jy,jm,jd}}
function toJ(iso){const [gy,gm,gd]=String(iso||nowKey()).slice(0,10).split('-').map(Number);return d2j(g2d(gy,gm,gd))}
function toG(jy,jm,jd){const g=d2g(j2d(Number(jy),Number(jm),Number(jd)));return `${g.gy}-${String(g.gm).padStart(2,'0')}-${String(g.gd).padStart(2,'0')}`}
function jDays(y,m){if(m<=6)return 31;if(m<=11)return 30;return jalCal(Number(y)).leap===0?30:29}
const JM=['فروردین','اردیبهشت','خرداد','تیر','مرداد','شهریور','مهر','آبان','آذر','دی','بهمن','اسفند'];
function jalaliPicker(prefix,iso=nowKey()){const j=toJ(iso),years=Array.from({length:8},(_,i)=>j.jy-2+i);return `<div class="jalaliDateSelects"><select id="${prefix}_y" onchange="v18FixJDay('${prefix}')">${years.map(y=>`<option value="${y}" ${y===j.jy?'selected':''}>${fmt(y)}</option>`).join('')}</select><select id="${prefix}_m" onchange="v18FixJDay('${prefix}')">${JM.map((m,i)=>`<option value="${i+1}" ${i+1===j.jm?'selected':''}>${m}</option>`).join('')}</select><select id="${prefix}_d">${Array.from({length:jDays(j.jy,j.jm)},(_,i)=>i+1).map(d=>`<option value="${d}" ${d===j.jd?'selected':''}>${fmt(d)}</option>`).join('')}</select></div>`}
window.v18FixJDay=function(p){const y=Number($('#'+p+'_y').value),m=Number($('#'+p+'_m').value),el=$('#'+p+'_d'),old=Number(el.value||1),n=jDays(y,m);el.innerHTML=Array.from({length:n},(_,i)=>`<option value="${i+1}" ${i+1===Math.min(old,n)?'selected':''}>${fmt(i+1)}</option>`).join('')};
